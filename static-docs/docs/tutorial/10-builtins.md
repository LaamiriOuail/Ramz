# الدوال المدمجة - Built-in Functions

في هذا الدرس، سنتعلم الدوال الجاهزة في لغة رمز!

## ما هي الدوال المدمجة؟

الدوال المدمجة هي دوال جاهزة يمكنك استخدامها فوراً دون تعريفها!

## الدوال الأساسية

### 1. اكتب() - الطباعة

تطبع النص على الشاشة:

```python
اكتب("مرحباً بالعالم!")
اكتب(42)
اكتب(3.14)
اكتب(صحيح)
```

### مثال متقدم:

```python
متغير الاسم = "أحمد"
متغير العمر = 15

اكتب("الاسم: " + الاسم)
اكتب("العمر: " + العمر)
اكتب("مكان: القاهرة")
```

### 2. اقرأ() - القراءة

تقرأ مدخلات من المستخدم:

```python
# قراءة بدون رسالة
متغير المدخلات = اقرأ()

# قراءة مع رسالة
متغير الاسم = اقرأ("ما اسمك؟ ")
```

### مثال كامل:

```python
اكتب("مرحباً بك في رمز!")
متغير الاسم = اقرأ("ما اسمك؟ ")
متغير العمر = اقرأ("كم عمرك؟ ")
متغير المدينة = اقرأ("من أين أنت؟ ")

اكتب("")
اكتب("=== معلوماتك ===")
اكتب("الاسم: " + الاسم)
اكتب("العمر: " + العمر + " سنة")
اكتب("المدينة: " + المدينة)
```

### 3. ادع() - استدعاء دالة

تستدعي دالة أخرى:

```python
دالة ترحيب() {
    ارجع "مرحباً بالعالم!"
}

اكتب(ادع ترحيب())
```

## مثال: آلة حاسبة بسيطة

```python
اكتب("=== آلة حاسبة بسيطة ===")

متغير ر1 = اقرأ("الرقم الأول: ")
متغير ر2 = اقرأ("الرقم الثاني: ")

اكتب("")
اكتب("العمليات:")
اكتب("1. جمع")
اكتب("2. طرح")
اكتب("3. ضرب")
اكتب("4. قسمة")

متغير العملية = اقرأ("اختر (1-4): ")

إذا العملية == "1":
    اكتب("النتيجة: " + (ر1 + r2))
وإلا إذا العملية == "2":
    اكتب("النتيجة: " + (r1 - r2))
وإلا إذا العملية == "3":
    اكتب("النتيجة: " + (r1 * r2))
وإلا إذا العملية == "4":
    اكتب("النتيجة: " + (r1 / r2))
وإلا:
    اكتب("❌ اختيار خاطئ!")
```

## أمثلة عملية

### مثال 1: برنامج الترحيب الذكي

```python
اكتب("=== برنامج الترحيب الذكي ===")

# قراءة المعلومات
متغير الاسم = اقرأ("ما اسمك؟ ")
متغير الوقت = اقرأ("ما الوقت الآن؟ (صباحاً/مساءاً): ")

# ترحيب حسب الوقت
اكتب("")
إذا الوقت == "صباحاً":
    اكتب("صباح الخير يا " + الاسم + "! ☀️")
وإلا إذا الوقت == "مساءاً":
    اكتب("مساء الخير يا " + الاسم + "! 🌙")
وإلا:
    اكتب("مرحباً " + الاسم + "! 👋")

# رسالة إضافية
اكتب("أهلاً وسهلاً بك في برنامج رمز! 🚀")
```

### مثال 2: تحويل درجة الحرارة

```python
اكتب("=== تحويل درجة الحرارة ===")

اكتب("1. مئوية إلى فهرنهايت")
اكتب("2. فهرنهايت إلى مئوية")

متغير الاختيار = اقرأ("اختر (1-2): ")

إذا الاختيار == "1":
    متغير مئوية = اقرأ("أدخل الدرجة المئوية: ")
    متغير فهرنهايت = مئوية * 1.8 + 32
    اكتب(mئوية + "°C = " + فهرنهايت + "°F")

وإلا إذا الاختيار == "2":
    متغير فهرنهايت = اقرأ("أدخل الدرجة الفهرنهايتية: ")
    متغير مئوية = (فهرنهايت - 32) / 1.8
    اكتب(فهرنهايت + "°F = " + مئوية + "°C")
```

### مثال 3: قائمة المهام التفاعلية

```python
اكتب("=== قائمة المهام ===")

# قائمة المهام
متغير المهام = []

# قائمة الخيارات
اكتب("")
اكتب("1. إضافة مهمة")
اكتب("2. عرض المهام")
اكتب("3. إنهاء")

متغير مستمر = صحيح
بينما مستمر:
    متغير الاختيار = اقرأ("اختر (1-3): ")

    إذا الاختيار == "1":
        متغير مهمة = اقرأ("أدخل المهمة: ")
        المهام = المهام + [مهمة]
        اكتب("✅ تمت الإضافة!")

    وإلا إذا الاختيار == "2":
        اكتب("")
        اكتب("=== قائمة المهام ===")
        إذا المهام == []:
            اكتب("لا توجد مهام")
        وإلا:
            متغير عداد = 1
            لكل مهمة في المهام:
                اكتب(عداد + ". " + مهمة)
                عداد = عداد + 1

    وإلا إذا الاختيار == "3":
        مستمر = خطأ
        اكتب("👋 مع السلامة!")

    وإلا:
        اكتب("❌ اختيار خاطئ!")

    اكتب("")
```

### مثال 4: لعبة التخمين المحسنة

```python
اكتب("=== لعبة التخمين ===")

متغير الرقم_السري = 42
متغير المحاولات = 0
متغير الحد_الأقصى = 10
متغير فوز = خطأ

اكتب("أنا فكرت في رقم بين 1 و 100")
اكتب("لديك " + الحد_الأقصى + " محاولات!")

بينما المحاولات < الحد_الأقصى و ليس فوز:
    متغير التخمين = اقرأ("تخمين #" + (المحاولات + 1) + ": ")

    المحاولات = المحاولات + 1

    إذا التخمين == الرقم_السري:
        فوز = صحيح
        اكتب("")
        اكتب("🎉🎉🎉 مبروك!")
        اكتب("لقد خمنت الرقم الصحيح!")
        اكتب("عدد المحاولات: " + المحاولات)

    وإلا إذا المحاولات == الحد_الأقصى:
        اكتب("")
        اكتب("❌ انتهت المحاولات!")
        اكتب("الرقم كان: " + الرقم_السري)

    وإلا إذا التخمين > الرقم_السري:
        متغير المتبقي = الحد_الأقصى - المحاولات
        اكتب("📉 كبير جداً! (متبقي: " + المتبقي + ")")

    وإلا:
        متغير المتبقي = الحد_الأقصى - المحاولات
        اكتب("📈 صغير جداً! (متبقي: " + المتبقي + ")")
```

### مثال 5: آلة الحاسبة العلمية

```python
اكتب("=== آلة الحاسبة العلمية ===")

متغير الرقم = اقرأ("أدخل رقماً: ")

اكتب("")
اكتب("النتائج:")
اكتب("-" * 30)
اكتب("العدد: " + الرقم)
اكتب("المربع: " + (الرقم * الرقم))
اكتب("المكعب: " + (الرقم * الرقم * الرقم))
اكتب("المضروب: " + (ادع مضروب(الرقم)))
اكتب("-" * 30)

دالة مضروب(عدد) {
    إذا عدد == 0 أو عدد == 1:
        ارجع 1
    و:
        ارجع عدد * ادع مضروب(عدد - 1)
}
```

### مثال 6: برنامج القائمة التفاعلي

```python
اكتب("=== برنامج القائمة ===")

# إنشاء قائمة فارغة
متغير القائمة = []

اكتب("")
اكتب("الخيارات:")
اكتب("1. إضافة")
اكتب("2. عرض")
اكتب("3. حذف")
اكتب("4. بحث")
اكتب("5. إنهاء")

متغير مستمر = صحيح

بينما مستمر:
    اكتب("")
    متغير الاختيار = اقرأ("اختر: ")

    إذا الاختيار == "1":
        # إضافة
        متغير عنصر = اقرأ("أدخل العنصر: ")
        القائمة = القائمة + [عنصر]
        اكتب("✅ تمت الإضافة!")

    وإلا إذا الاختيار == "2":
        # عرض
        اكتب("")
        اكتب("=== محتويات القائمة ===")
        متغير عداد = 1
        لكل عنصر في القائمة:
            اكتب(عداد + ". " + عنصر)
            عداد = عداد + 1

    وإلا إذا الاختيار == "3":
        # حذف
        متغير موقع = اقرأ("رقم العنصر لحذفه: ")
        القائمة[mوقع] = ""
        اكتب("✅ تم الحذف!")

    وإلا إذا الاختيار == "4":
        # بحث
        متغير بحث = اقرأ("ابحث عن: ")
        متغير موجود = خطأ
        لكل عنصر في القائمة:
            إذا عنصر == بحث:
                موجود = صحيح
                اوقف

        إذا موجود:
            اكتب("✅ موجود!")
        وإلا:
            اكتب("❌ غير موجود!")

    وإلا إذا الاختيار == "5":
        مستمر = خطأ
        اكتب("👋 مع السلامة!")

    وإلا:
        اكتب("❌ اختيار خاطئ!")
```

## نصائح مهمة

💡 **استخدم دالة اقرأ() بشكل ذكي**
```python
# ✅ استخدم رسالة واضحة
متغير الاسم = اقرأ("ما اسمك؟ ")

# ❌ رسالة غير واضحة
متغير الاسم = اقرأ()
```

💡 **طالب بمعلومات متدرجة**
```python
# ✅ معلومات واضحة ومتدرجة
متغير الاسم = اقرأ("الاسم: ")
متغير العمر = اقرأ("العمر: ")
متغير المدينة = اقرأ("المدينة: ")

# ❌ غير واضح
متغير الاسم = اقرأ()
متغير العمر = اقرأ()
```

💡 **استخدم اكتب() لتوضيح ما يحدث**
```python
# ✅ واضح
اكتب("جاري الحساب...")
اكتب("النتيجة: " + النتيجة)

# ❌ غامض
اكتب(النتيجة)
```

## ملخص

✅ تعلمت الدالة `اكتب()` للطباعة
✅ فهمت الدالة `اقرأ()` للقراءة من المستخدم
✅ عرفت الدالة `ادع()` لاستدعاء دوال أخرى
✅ أنشأت برامج تفاعلية مع المستخدم
✅ استخدمت الدوال مع الشرطيات والحلقات
✅ بنيت برامج تفاعلية كاملة

## جدول الدوال المدمجة

| الدالة | الاستخدام | مثال |
|-------|----------|-------|
| `اكتب()` | طباعة على الشاشة | `اكتب("مرحباً")` |
| `اقرأ()` | قراءة من المستخدم | `اقرأ("اسمك؟ ")` |
| `ادع()` | استدعاء دالة | `ادع ترحيب()` |

في الدرس القادم، سنتعلم كيفية التعامل مع الأخطاء والتحقق من صحة البيانات!
