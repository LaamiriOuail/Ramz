# مرجع الدوال - Functions Reference

دليل سريع لاستخدام الدوال في لغة رمز.

## البنية الأساسية

### تعريف دالة

```python
دالة اسم_الدالة(المعامل1, المعامل2) {
    # الكود هنا
    ارجع قيمة
}
```

### استدعاء دالة

```python
ادع اسم_الدالة(القيمة1, القيمة2)
```

## الكلمات المفتاحية للدوال

| الكلمة | الاستخدام | مثال |
|--------|-----------|------|
| `دالة` | تعريف دالة جديدة | `دالة جمع()` |
| `ارجع` | إرجاع قيمة من الدالة | `ارجع 42` |
| `ادع` | استدعاء دالة | `ادع جمع(1, 2)` |

## أمثلة الأساسية

### دالة بدون معاملات

```python
دالة مرحب() {
    ارجع "مرحباً"
}

اكتب(ادع مرحب())
```

### دالة مع معامل واحد

```python
دالة مضاعف(عدد) {
    ارجع عدد * 2
}

اكتب(ادع مضاعف(5))
```

### دالة مع عدة معاملات

```python
دالة جمع(أ, ب) {
    ارجع أ + ب
}

اكتب(ادع جمع(10, 20))
```

## أمثلة عملية

### العمليات الحسابية

```python
دالة جمع(أ, ب) { ارجع أ + ب }
دالة طرح(أ, ب) { ارجع أ - ب }
دالة ضرب(أ, ب) { ارجع أ * ب }
دالة قسمة(أ, ب) { ارجع أ / ب }
```

### التعامل مع النصوص

```python
دالة تحية(الاسم) {
    ارجع "مرحباً " + الاسم + "!"
}

دالة كرر_النص(النص, عدد) {
    متغير النتيجة = ""
    متغير عداد = 0
    بينما عداد \`<\` عدد {
        النتيجة = النتيجة + النص
        عداد = عداد + 1
    }
    ارجع النتيجة
}
```

### الحسابات الهندسية

```python
دالة مساحة_مربع(الطول) {
    ارجع الطول * الطول
}

دالة مساحة_مستطيل(الطول, العرض) {
    ارجع الطول * العرض
}

دالة مساحة_مثلث(القاعدة, الارتفاع) {
    ارجع (القاعدة * الارتفاع) / 2
}

دالة مساحة_دائرة(نصف_القطر) {
    ارجع نصف_القطر * نصف_القطر * 3.14
}
```

### دوال متداخلة (دوال تستدعي دوال أخرى)

```python
دالة مربع(عدد) {
    ارجع عدد * عدد
}

دالة مكعب(عدد) {
    ارجع ادع مربع(عدد) * عدد
}

دالة مرفوع_للأس(عدد, الأس) {
    متغير النتيجة = 1
    متغير عداد = 0
    بينما عداد < الأس {
        النتيجة = النتيجة * عدد
        عداد = عداد + 1
    }
    ارجع النتيجة
}
```

### دوال تحويلية

```python
دالة مئوية_إلى_فهرنهايت(مئوية) {
    ارجع مئوية * 1.8 + 32
}

دالة فهرنهايت_إلى_مئوية(فهرنهايت) {
    ارجع (فهرنهايت - 32) / 1.8
}

دالة كيلومتر_إلى_ميل(كم) {
    ارجع كم * 0.621371
}
```

### دوال للتحقق

```python
دالة عدد_زوجي(عدد) {
    ارجع عدد % 2 == 0
}

دالة عدد_فردي(عدد) {
    ارجع عدد % 2 != 0
}

دالة إيجابي(عدد) {
    ارجع عدد > 0
}

دالة سالب(عدد) {
    ارجع عدد < 0
}
```

### دوال للقوائم

```python
دالة مجموع_قائمة(قائمة) {
    متغير المجموع = 0
    لكل رقم في قائمة {
        المجموع = المجموع + رقم
    }
    ارجع المجموع
}

دالة أقصى(قائمة) {
    متغير الأقصى = قائمة[0]
    لكل رقم في قائمة {
        إذا رقم > الأقصى {
            الأقصى = رقم
        }
    }
    ارجع الأقصى
}
```

## قواعد مهمة

### ✅ أمثلة صحيحة

```python
# اسم دالة واضح
دالة حساب_المسافة(س1, ص1, س2, ص2) { }

# معاملات مع أسماء واضحة
دالة مستطيل(الطول, العرض) { }

# دالة تُرجع قيمة واحدة
دالة ضع_في_المربع(عدد) {
    ارجع عدد * عدد
}
```

### ❌ أمثلة خاطئة

```python
# اسم دالة غير واضح
دالة x(a, b) { }

# دالة بدون إرجاع قيمة (لن تفيد)
دالة مجموع(أ, ب) {
    أ + ب
}

# دالة تقوم بأكثر من وظيفة
دالة كل_شيء(أ, ب) {
    ارجع أ + ب
    اكتب("مرحباً")
}
```

## نصائح البرمجة

### 1. وظيفة واحدة لكل دالة
```python
# ✅ جيد
دالة مربع(عدد) { ارجع عدد * عدد }
دالة مكعب(عدد) { ارجع ادع مربع(عدد) * عدد }

# ❌ سيء
دالة مربع_ومكعب(عدد) {
    ارجع عدد * عدد
    ارجع عدد * عدد * عدد
}
```

### 2. أسماء واضحة
```python
# ✅ جيد
دالة حساب_الضريبة(السعر) { }
دالة تحويل_العملة(المبلغ) { }

# ❌ سيء
دالة f(x) { }
دالة do_it(a, b) { }
```

### 3. تكرار الكود = دالة جديدة
```python
# إذا كررت هذا الكود مرات، اجعله دالة!
متغير مجموع = a + b
متغير مجموع = c + d
متغير مجموع = e + f

# أفضل:
دالة جمع(أ, ب) { ارجع أ + ب }
ادع جمع(a, b)
ادع جمع(c, d)
ادع جمع(e, f)
```

## أمثلة متقدمة

### دالة المضروب (Factorial)

```python
دالة مضروب(عدد) {
    إذا عدد == 0 أو عدد == 1 {
        ارجع 1
    }
    ارجع عدد * ادع مضروب(عدد - 1)
}
```

### دالة التحقق من الأعداد الأولية

```python
دالة أولي(عدد) {
    إذا عدد < 2 {
        ارجع خطأ
    }
    متغير قاسم = 2
    بينما قاسم \`<\` عدد {
        إذا عدد % قاسم == 0 {
            ارجع خطأ
        }
        قاسم = قاسم + 1
    }
    ارجع صحيح
}
```
