# الأنواع في رمز - Types

في هذا الدرس، سنتعلم كل شيء عن أنواع البيانات في لغة رمز!

## ما هي الأنواع؟

الأنواع تخبرنا نوع البيانات التي نتعامل معها:
- رقم أم نص؟
- عدد صحيح أم عشري؟
- قائمة أم قاموس؟

## الأنواع الأساسية

### 1. رقم (Number) - الأعداد الصحيحة

```python
متغير العمر = 15
متغير السنة = 2024
متغير الكمية = 100
```

### 2. عشري (Float) - الأعداد العشرية

```python
متغير الطول = 1.75
متغير الوزن = 65.5
متغير السعر = 99.99
متغير نسبة = 0.5
```

### 3. نص (String) - النصوص

```python
متغير الاسم = "أحمد"
متغير المدينة = "القاهرة"
متغير الرسالة = "مرحباً بالعالم!"
متغير الرمز = "@#$%"
```

### 4. منطقية (Boolean) - القيم المنطقية

```python
متغير سعيد = صحيح
متغير متعب = خطأ
متغير نشط = صحيح
متغير فارغ = خطأ
```

## الأنواع المتقدمة

### 5. قائمة (List)

```python
متغير الأرقام = [1, 2, 3, 4, 5]
متغير الأسماء = ["أحمد", "محمد", "خالد"]
متغير مختلط = [1, "نص", 3.14, صحيح]
```

### 6. قاموس (Dictionary)

```python
متغير الطالب = {
    "الاسم": "أحمد",
    "العمر": 15,
    "المدينة": "القاهرة"
}

متغير الأسعار = {
    "تفاح": 5,
    "موز": 3,
    "برتقال": 4
}
```

### 7. زوج (Tuple)

```python
متغير الإحداثيات = (10, 20)
متغير التاريخ = (2024, 3, 15)
متغير اللون = (255, 128, 0)
```

## تحديد النوع صراحة

يمكنك تحديد نوع المتغير عند إنشائه:

### تحديد الأنواع الأساسية

```python
# رقم صحيح
متغير العمر: رقم = 15

# رقم عشري
متغير الطول: عشري = 1.75

# نص
متغير الاسم: نص = "أحمد"

# منطقية
متغير نشط: منطقية = صحيح

# قائمة
متغير الأرقام: قائمة = [1, 2, 3, 4, 5]

# قاموس
متغير الطالب: قاموس = {"الاسم": "أحمد"}

# زوج
متغير الإحداثيات: زوج = (10, 20)
```

## لماذا نحدد الأنواع؟

### الفائدة 1: وضوح الكود

```python
# دون تحديد النوع - غير واضح
متغير x = 15

# مع تحديد النوع - واضح جداً
متغير العمر: رقم = 15
```

### الفائدة 2: منع الأخطاء

```python
# سيخطئ إذا حاولت وضع نص
متغير العمر: رقم = "خمسة عشر"  # ❌ خطأ!

# صحيح
متغير العمر: رقم = 15  # ✅ جيد
```

### الفائدة 3: تحسين الأداء

```python
# النوع المحدد يجعل البرنامج أسرع
متغير العدد: رقم = 1000000
```

## التحقق من النوع

يمكنك معرفة نوع المتغير:

```python
متغير العمر = 15
متغير الاسم = "أحمد"

# يمكنك استخدام المتغيرات في عمليات مختلفة
اكتب(العمر + 1)        # 16
اكتب(الاسم + "!")       # أحمد!
```

## التحويل بين الأنواع

### من نص إلى رقم

```python
متغير نص_الرقم = "42"
متغير الرقم = رقم(نص_الرقم)  # 42
```

### من رقم إلى نص

```python
متغير العمر = 15
متغير نص_العمر = نص(العمر)  # "15"

اكتب("عمري " + نص(العمر) + " سنة")
```

### مثال عملي:

```python
متغير العمر_نص = اقرأ("كم عمرك؟ ")
اكتب("نص: " + العمر_نص)
اكتب("رقم: " + العمر_نص + 1)
```

## العمليات حسب النوع

### عمليات الأرقام

```python
متغير أ = 10
متغير ب = 3

اكتب(أ + ب)   # 13 (جمع)
اكتب(أ - ب)   # 7 (طرح)
اكتب(أ * ب)   # 30 (ضرب)
اكتب(أ / ب)   # 3.33 (قسمة)
اكتب(أ % ب)   # 1 (باقي القسمة)
```

### عمليات النصوص

```python
متغير نص1 = "مرحباً"
متغير نص2 = " بالعالم"

اكتب(نص1 + نص2)  # مرحباً بالعالم
```

### عمليات القوائم

```python
متغير قائمة = [1, 2, 3, 4, 5]

# الوصول لعنصر
اكتب(قائمة[0])  # 1
اكتب(قائمة[2])  # 3

# إضافة عنصر
قائمة = قائمة + [6, 7]
```

## مثال شامل

### مثال: إدارة طالب

```python
# تعريف متغيرات بأنواع مختلفة
متغير اسم_الطالب: نص = "أحمد محمد"
متغير العمر: رقم = 15
متغير الطول: عشري = 1.65
متغير نشط: منطقية = صحيح
متغير الدرجات: قائمة = [85, 92, 78, 88, 95]
متغير المعلومات: قاموس = {
    "المدينة": "القاهرة",
    "المدرسة": "الابتدائية الأولى",
    "الصف": "الثالث"
}

# طباعة المعلومات
اكتب("=== معلومات الطالب ===")
اكتب("الاسم: " + اسم_الطالب)
اكتب("العمر: " + العمر + " سنة")
اكتب("الطول: " + الطول + " متر")
اكتب("نشط: " + نشط)
اكتب("الدرجات: " + الدرجات)
اكتب("المدينة: " + المعلومات["المدينة"])
```

## أمثلة عملية

### مثال 1: آلة حاسبة مع أنواع

```python
اكتب("=== آلة حاسبة ذكية ===")

# قراءة كنصوص
متغير ر1: نص = اقرأ("الرقم الأول: ")
متغير ر2: نص = اقرأ("الرقم الثاني: ")

# سيتم تحويلها تلقائياً عند الحساب
متغير مجموع = ر1 + ر2
متغير فرق = r1 - r2

اكتب("المجموع: " + مجموع)
اكتب("الفرق: " + فرق)
```

### مثال 2: قائمة مهام

```python
# قائمة مهام (نصوص)
متغير المهام: قائمة = [
    "إنجاز الواجب",
    "مراجعة الدرس",
    "القراءة لمدة ساعة"
]

اكتب("=== قائمة المهام ===")
متغير عداد = 1
لكل مهمة في المهام:
    اكتب(عداد + ". " + مهمة)
    عداد = عداد + 1
```

### مثال 3: معلومات منتج

```python
# قاموس معلومات منتج
متغير المنتج: قاموس = {
    "الاسم": "لابتوب",
    "السعر": 5000,
    "متوفر": صحيح,
    "المراجعات": [5, 4, 5, 5, 4]
}

# عرض المعلومات
اكتب("=== معلومات المنتج ===")
اكتب("الاسم: " + المنتج["الاسم"])
اكتب("السعر: " + المنتج["السعر"] + " ريال")
اكتب("متوفر: " + المنتج["متوفر"])
اكتب("المراجعات: " + المنتج["المراجعات"])
```

### مثال 4: تاريخ وأوقات

```python
# زوج للإحداثيات
متغير الموقع: زوج = (30.0444, 31.2357)
اكتب("الموقع: " + الموقع)

# زوج للتاريخ
متغير التاريخ: زوج = (2024, 3)
اكتب("التاريخ: " + التاريخ)
```

## تمرينات

### تمرين 1: إنشاء متغيرات بأنواع مختلفة

```python
# أنشئ المتغيرات التالية
متغير اسمي: نص = "اسمك"
متغير عمري: رقم = 15
متغير طوالي: عشري = 1.60
متغير طالب: منطقية = صحيح
متغير درجاتي: قائمة = [80, 85, 90]
متغير بياناتي: قاموس = {"مدينتي": "القاهرة"}

# اطبعها
اكتب(اسمي)
اكتب(عمري)
اكتب(طوالي)
اكتب(طالب)
اكتب(درجاتي)
اكتب(بياناتي["مدينتي"])
```

### تمرين 2: قائمة أرقام

```python
# قائمة أرقام
متغير الأرقام: قائمة = [10, 20, 30, 40, 50]

# احسب المجموع
متغير المجموع = 0
لكل رقم في الأرقام:
    المجموع = المجموع + رقم

اكتب("المجموع: " + المجموع)
```

### تمرين 3: قاموس كتاب

```python
# معلومات كتاب
متغير الكتاب: قاموس = {
    "العنوان": "مغامرات رمز",
    "المؤلف": "فريق رمز",
    "السنة": 2024,
    "متوفر": صحيح
}

# اطبع المعلومات
اكتب("=== معلومات الكتاب ===")
اكتب("العنوان: " + الكتاب["العنوان"])
اكتب("المؤلف: " + الكتاب["المؤلف"])
اكتب("السنة: " + الكتاب["السنة"])
اكتب("متوفر: " + الكتاب["متوفر"])
```

## نصائح مهمة

💡 **اختر النوع المناسب**
```python
# ✅ صحيح
متغير العمر: رقم = 15

# ❌ غير ضروري (العمر لا يحتاج كسور)
متغير العمر: عشري = 15.0
```

💡 **استخدم الأنواع لتوضيح الكود**
```python
# ✅ واضح
متغير الأسعار: قائمة = [10, 20, 30]
متغير معلومات_المنتج: قاموس = {"الاسم": "لابتوب"}

# ❌ غير واضح
متغير x = [10, 20, 30]
متغير y = {"الاسم": "لابتوب"}
```

💡 **كن متسقاً**
```python
# ✅ استخدام متسق للأنواع
متغير الأسماء: قائمة = []
متغير الأرقام: قائمة = []

# ❌ غير متسق
متغير x = []
متغير y: قائمة = []
```

✅ عرفت كيفية تحديد النوع صراحة
✅ فهمت فائدة تحديد الأنواع
✅ تعلمت التحويل بين الأنواع
✅ استخدمت الأنواع في أمثلة عملية
✅ عرفت العمليات المختلفة لكل نوع
